import { createClient } from "npm:@supabase/supabase-js@2.45.4";

const db = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);
const KEY = (Deno.env.get("ANTHROPIC_API_KEY") || "").trim();
const API_VERSION = Deno.env.get("BL_API_VERSION") || "2026-10-21";
const AGENT_KEY = Deno.env.get("AGENT_KEY") || "";
const MODEL = Deno.env.get("BL_MODEL") || "claude-sonnet-5-5";
const SKIP = (Deno.env.get("BL_SKIP") || "paylink").split(",").map((s) => s.trim()).filter(Boolean);

const parentOf = (p: string) => p.split("/").slice(0, -1).join("/");

async function getDoc(path: string) {
  const { data } = await db.from("docs").select("data").eq("path", path).maybeSingle();
  return data ? data.data : null;
}
async function listCol(path: string) {
  const { data } = await db.from("docs").select("path,data").eq("parent", path);
  return (data || []).map((r: any) => ({ id: r.path.split("/").pop(), ...r.data }));
}
async function put(path: string, data: unknown) {
  const { error } = await db.from("docs").upsert({ path, parent: parentOf(path), data, updated_at: new Date().toISOString() });
  if (error) throw error;
}
async function merge(path: string, patch: Record<string, unknown>) {
  const cur = (await getDoc(path)) || {};
  await put(path, { ...cur, ...patch }); // conserva el campo cifrado "e" tal cual
}

async function claude(prompt: string) {
  const body = JSON.stringify({
    model: MODEL,
    max_tokens: 16000,
    tools: [{ type: "web_search_20250305", name: "web_search", max_uses: 10 }],
    messages: [{ role: "user", content: prompt }],
  });
  const base = { "anthropic-version": API_VERSION, "content-type": "application/json" };
  let r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { ...base, "x-api-key": KEY }, body });
  if (r.status === 401) {
    // algunas claves nuevas se mandan como "Authorization: Bearer"
    r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { ...base, authorization: `Bearer ${KEY}` }, body });
  }
  const d = await r.json();
  if (!r.ok) throw new Error(`${d?.error?.message || "Anthropic"} (${r.status}; clave ${KEY.slice(0, 10)}…, ${KEY.length} caracteres)`);
  const text = (d.content || []).filter((b: any) => b.type === "text").map((b: any) => b.text).join("\n");
  const m = text.match(/\{[\s\S]*\}$/m) || text.match(/\{[\s\S]*\}/);
  if (!m) throw new Error("La respuesta no trajo JSON");
  try {
    return JSON.parse(m[0]);
  } catch {
    // a veces vienen saltos de línea dentro de los textos: se limpian y se intenta otra vez
    return JSON.parse(m[0].replace(/[\u0000-\u001F]+/g, " "));
  }
}

function brief(c: any) {
  return `Marca: ${c.name}\nSector: ${c.industry || "-"}\nPaís de su audiencia: ${c.country || "Guatemala"}\nSitio web: ${c.website || "-"}\nAudiencia: ${c.audience || "-"}\nNotas: ${String(c.notes || "").slice(0, 1200)}`;
}

async function research(cid: string) {
  const c = await clientOf(cid);
  if (!c || !c.name) throw new Error("marca sin datos públicos");
  const comps = await listCol(`clients/${cid}/competitors`);
  const events = await listCol(`clients/${cid}/events`);
  const today = new Date().toISOString().slice(0, 10);
  const prompt = `Eres el agente de investigación de Brandloop, una plataforma de marketing en Guatemala. Hoy es ${today}.
${brief(c)}

Investiga en la web (varias búsquedas, prioriza lo publicado en los últimos 14 días y fuentes de su país y región). Abre su sitio web para entender qué vende.
Competidores que ya tenemos (no los repitas como nuevos): ${comps.map((x: any) => x.name).join("; ") || "ninguno"}.
Eventos que ya tenemos: ${events.map((x: any) => x.name).join("; ") || "ninguno"}.

Reglas: no inventes datos, cifras, eventos ni URLs; todo debe venir de una fuente que hayas visto. Parafrasea. Español.
Responde SOLO con un JSON con esta forma:
{"sector":[{"title","summary","why","idea","source","url"}x4],
 "social":[{"title","summary","why","idea","source","url","platform"}x4],
 "tags":{"es":[8-15 hashtags vistos en su nicho],"en":[8-15]},
 "newCompetitors":[{"name","type":"directa|indirecta","url","country","does":[2-4],"doesnt":[2-4],"channels":["Instagram","Facebook","LinkedIn","TikTok","YouTube","Web"],"channelsNote","opportunity"}] (0-4, solo importantes y reales),
 "events":[{"name","type","date":"YYYY-MM-DD","end","date_confirmed":true|false,"city","country","url","audience","why","play","cost"}] (próximos 12 meses, 0-6, solo nuevos o con datos actualizados),
 "summary":"1 frase con lo más importante para la marca esta semana"}`;
  const r = await claude(prompt);
  const t = now();
  const writes: Promise<unknown>[] = [];
  (r.sector || []).slice(0, 4).forEach((x: any, i: number) => writes.push(put(`clients/${cid}/trends/s${i + 1}`, { ...x, kind: "sector", order: i + 1, updatedAt: t })));
  (r.social || []).slice(0, 4).forEach((x: any, i: number) => writes.push(put(`clients/${cid}/trends/r${i + 1}`, { ...x, kind: "social", order: 10 + i, updatedAt: t })));
  if (r.tags) writes.push(merge(`clients/${cid}`, { trendTags: r.tags }));
  let k = comps.reduce((m: number, x: any) => Math.max(m, Number(String(x.id).replace(/\D/g, "")) || 0), 0);
  const have = new Set(comps.map((x: any) => String(x.name || "").toLowerCase().replace(/[^a-z0-9]/g, "")));
  for (const x of (r.newCompetitors || []).slice(0, 4)) {
    const key = String(x.name || "").toLowerCase().replace(/[^a-z0-9]/g, "");
    if (!key || have.has(key)) continue;
    have.add(key);
    writes.push(put(`clients/${cid}/competitors/k${++k}`, { ...x, order: 100 + k, updatedAt: t, source: "investigacion" }));
  }
  let e = events.reduce((m: number, x: any) => Math.max(m, Number(String(x.id).replace(/\D/g, "")) || 0), 0);
  for (const x of (r.events || []).slice(0, 6)) {
    const old = events.find((y: any) => String(y.name || "").toLowerCase() === String(x.name || "").toLowerCase());
    if (old && old.source === "manual") continue;
    const path = old ? `clients/${cid}/events/${old.id}` : `clients/${cid}/events/e${++e}`;
    writes.push(put(path, { ...(old || {}), ...x, status: old?.status, estCost: old?.estCost, order: old?.order ?? e, updatedAt: t, source: "investigacion" }));
  }
  for (const x of events) if (x.date && x.date < today && !x.status && x.source !== "manual") writes.push(put(`clients/${cid}/events/${x.id}`, { ...x, status: "pasado" }));
  writes.push(put(`clients/${cid}/docs/research`, { at: t, summary: r.summary || "", model: MODEL }));
  await Promise.all(writes);
  return { cid, summary: r.summary || "" };
}


// ---------- Llaves: el agente puede leer y guardar datos privados de cada marca, cifrados ----------
// La primera vez el agente crea su par de llaves. La privada queda en la tabla bl_agent_secret (sin acceso
// desde la app ni desde internet; solo el servidor). La pública se publica en config/agent; la app la usa
// para entregarle al agente la llave de cada marca (vault/agentkeys), cifrada para que solo el agente la abra.
const te = new TextEncoder(), td = new TextDecoder();
const b64 = (u: Uint8Array) => { let s = ""; for (let i = 0; i < u.length; i += 0x8000) s += String.fromCharCode(...u.subarray(i, i + 0x8000)); return btoa(s); };
const ub64 = (s: string) => Uint8Array.from(atob(s), (c) => c.charCodeAt(0));
const RSA = { name: "RSA-OAEP", hash: "SHA-256" };
const SECRET_COLS = new Set(["posts", "people", "lposts", "tasks", "deals", "metrics", "plan", "campaigns", "ads", "geo", "organic", "emailcamps"]);
const SECRET_DOCS = new Set(["finance", "adcalc", "pautastrat", "funnel", "compass", "prodstories", "investplan", "strategy", "audience", "innovation", "site", "quote"]);
let PRIV: CryptoKey | null = null;
const BKEYS: Record<string, CryptoKey | null> = {};

async function agentPriv() {
  if (PRIV) return PRIV;
  const { data } = await db.from("bl_agent_secret").select("data").eq("id", "rsa").maybeSingle();
  let jwk = data?.data?.priv;
  if (!jwk) {
    const kp = await crypto.subtle.generateKey({ ...RSA, modulusLength: 3072, publicExponent: new Uint8Array([1, 0, 1]) }, true, ["encrypt", "decrypt"]) as CryptoKeyPair;
    jwk = await crypto.subtle.exportKey("jwk", kp.privateKey);
    const pub = await crypto.subtle.exportKey("jwk", kp.publicKey);
    const kid = String(Date.now());
    const { error } = await db.from("bl_agent_secret").insert({ id: "rsa", data: { priv: jwk, kid } });
    if (error) throw new Error("No se pudo guardar la llave del agente: " + error.message);
    await put("config/agent", { pub: { kty: pub.kty, n: pub.n, e: pub.e, alg: "RSA-OAEP-256" }, kid, at: now() });
  }
  PRIV = await crypto.subtle.importKey("jwk", jwk, RSA, false, ["decrypt"]);
  return PRIV;
}
async function brandKey(cid: string) {
  if (cid in BKEYS) return BKEYS[cid];
  const ak = (await getDoc("vault/agentkeys")) || {};
  const w = ak[cid]?.w;
  if (!w) return (BKEYS[cid] = null);
  const raw = new Uint8Array(await crypto.subtle.decrypt(RSA, await agentPriv(), ub64(w)));
  return (BKEYS[cid] = await crypto.subtle.importKey("raw", raw, { name: "AES-GCM" }, false, ["encrypt", "decrypt"]));
}
async function encJSON(k: CryptoKey, o: unknown, aad: string) {
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const ct = new Uint8Array(await crypto.subtle.encrypt({ name: "AES-GCM", iv, additionalData: te.encode(aad) }, k, te.encode(JSON.stringify(o))));
  const out = new Uint8Array(12 + ct.length); out.set(iv); out.set(ct, 12); return b64(out);
}
async function decJSON(k: CryptoKey, s: string, aad: string) {
  const u = ub64(s);
  return JSON.parse(td.decode(await crypto.subtle.decrypt({ name: "AES-GCM", iv: u.slice(0, 12), additionalData: te.encode(aad) }, k, u.slice(12))));
}
function aadOf(path: string) {
  const s = path.split("/");
  if (s[0] !== "clients" || !s[1]) return null;
  if (s.length === 2) return { kind: "client", aad: "clients/" + s[1] };
  if (SECRET_COLS.has(s[2])) return { kind: "doc", aad: `clients/${s[1]}/${s[2]}` };
  if (s[2] === "docs" && s[3] && SECRET_DOCS.has(s[3])) return { kind: "doc", aad: `clients/${s[1]}/docs` };
  return null;
}
// lee un documento y, si es privado, lo descifra con la llave de SU marca (nunca la de otra)
async function readPriv(path: string) {
  const d = await getDoc(path);
  const a = aadOf(path);
  if (!d || !a || !d.e) return d;
  const k = await brandKey(path.split("/")[1]);
  if (!k) return a.kind === "client" ? d : null;
  const p = await decJSON(k, d.e, a.aad);
  if (a.kind === "client") { const pub = { ...d }; delete pub.e; return { ...pub, ...p }; }
  return p;
}
// guarda un documento privado ya cifrado; si la marca no le ha dado su llave al agente, no guarda nada
async function writePriv(path: string, data: any) {
  const a = aadOf(path);
  if (!a || a.kind !== "doc") return put(path, data);
  const k = await brandKey(path.split("/")[1]);
  if (!k) throw new Error("sin llave de la marca");
  const out: any = { e: await encJSON(k, data, a.aad) };
  if (data?.order != null) out.order = data.order;
  if (data?.createdAt != null) out.createdAt = data.createdAt;
  return put(path, out);
}
async function clientOf(cid: string) {
  return (await readPriv(`clients/${cid}`)) || null;
}

function gtNow() { return new Date(Date.now() - 6 * 3600e3); }
function monthKey() { const d = gtNow(); return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`; }
function quarterGT() {
  const d = gtNow();
  const q = Math.floor(d.getUTCMonth() / 3), y = d.getUTCFullYear();
  const M = ["ene", "feb", "mar", "abr", "may", "jun", "jul", "ago", "sept", "oct", "nov", "dic"];
  return { label: `T${q + 1} ${y}`, months: [0, 1, 2].map((i) => M[q * 3 + i]), range: `${M[q * 3]}–${M[q * 3 + 2]} ${y}` };
}

    // primera vez: lo que ya existe cuenta como "al día" este mes; se actualiza a partir del próximo mes
    const init: Record<string, string> = {};
    for (const k of PLAN_ORDER) if (k !== "associations" && (await hasRow(`clients/${cid}/docs/${k}`))) init[k] = mk;
    await put(`clients/${cid}/docs/research`, { ...res, plan: init });
    res.plan = init;
  }
  const done = res.plan;
  const qStart = [0, 3, 6, 9].includes(gtNow().getUTCMonth());
  for (const k of PLAN_ORDER) {
    if (k === "associations") {
      const { count } = await db.from("docs").select("path", { count: "exact", head: true }).eq("parent", `clients/${cid}/associations`);
      if (!count && !done.associations) return { k, kind: "new" };
      continue;
    }
    if (!(await hasRow(`clients/${cid}/docs/${k}`))) return { k, kind: "new" };
  }
  for (const k of PLAN_ORDER) {
    if (k === "associations") continue;
    const last = String(done[k] || "");
    if (MONTHLY.has(k) && last < mk) return { k, kind: "update" };
    if (QUARTERLY.has(k) && qStart && last < mk) return { k, kind: "update" };
  }
  return null;
}

async function markDone(cid: string, k: string) {
  const res = (await getDoc(`clients/${cid}/docs/research`)) || {};
  await put(`clients/${cid}/docs/research`, { ...res, plan: { ...(res.plan || {}), [k]: monthKey() }, planError: null });
}

async function planStep(cid: string, job: { k: string; kind: string }) {
  if (job.kind === "research") return { step: "research", ...(await research(cid)) };
  const c = await clientOf(cid);
  if (!c || !c.name) throw new Error("marca sin datos públicos");
  const k = job.k;
  const path = `clients/${cid}/docs/${k}`;
  const cur = job.kind === "update" ? await readPriv(path) : null;
  if (job.kind === "update" && !cur) { await markDone(cid, k); return { cid, step: k, skipped: "no se pudo leer la versión vigente" }; }
  const base = cur ? (() => { const x = { ...cur }; delete x._prev; return x; })() : null;
  const r = await claude(planPrompt(k, c, await ctxOf(cid, c), base, MES[gtNow().getUTCMonth()]));
  const t = now();
  if (k === "webaudit") {
    await put(path, { ...r, status: "listo", url: r.url || c.website || "", updatedAt: t, source: "agente", _prev: base || undefined });
  } else if (k === "associations") {
    const items = (r.items || []).filter((x: any) => x && x.name).slice(0, 8);
    await Promise.all(items.map((x: any, i: number) => put(`clients/${cid}/associations/a${i + 1}`, { ...x, order: i + 1, updatedAt: t, source: "agente" })));
  } else {
    if (k === "strategy") {
      const Q = quarterGT();
      const sameQ = base && base.quarter === Q.label;
      const prog: Record<string, number> = {};
      if (sameQ) for (const x of base.strategies || []) prog[String(x.title || "").toLowerCase()] = Number(x.progress) || 0;
      Object.assign(r, { quarter: Q.label, range: Q.range, months: Q.months, strategies: (r.strategies || []).map((x: any) => ({ ...x, progress: prog[String(x.title || "").toLowerCase()] ?? (Number(x.progress) || 0) })) });
    }
    await writePriv(path, { ...r, updatedAt: t, source: "agente", _prev: base ? { ...base, savedAt: t } : undefined });
  }
  await markDone(cid, k);
  return { cid, step: k, kind: job.kind };
}

async function win(connector: string, accounts: string[], preset: string, fields: string[]) {

  if (!r.ok || d.error) throw new Error(`Windsor ${connector}: ${String(d.error || r.status).slice(0, 160)}`);

}
